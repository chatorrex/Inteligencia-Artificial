// Invertir cadena
void invertirCadena(char cadena[], char resultado[]) {
    int longitud = strlen(cadena);

    for (int i = 0; i < longitud; i++) {
        resultado[i] = cadena[longitud - 1 - i];
    }

    resultado[longitud] = '\0';
}

// Verificar palíndroma
bool verificarPalindroma(char cadena[]) {
    int inicio = 0;
    int fin = strlen(cadena) - 1;

    while (inicio < fin) {
        if (cadena[inicio] != cadena[fin]) {
            return false;
        }

        inicio++;
        fin--;
    }

    return true;
}

// 52 - Factorial
long long factorial(int num) {
    long long res = 1;

    for (int i = 1; i <= num; i++) {
        res *= i;
    }

    return res;
}

// 53 - Invertir número de izquierda a derecha
int invertirNumero(int num) {
    int res = 0;
    int aux;

    while (num > 0) {
        aux = num % 10;
        res = (res * 10) + aux;
        num /= 10;
    }

    return res;
}

// 55 - Binario a decimal
int binarioDecimal(int num) {
    int res = 0;
    int potencia = 1;
    int aux;

    while (num > 0) {
        aux = num % 10;
        res += aux * potencia;
        potencia *= 2;
        num /= 10;
    }

    return res;
}

// 56 - Decimal a binario
long long decimalBinario(int num) {
    long long res = 0;
    long long posicion = 1;
    int aux;

    while (num > 0) {
        aux = num % 2;
        res += aux * posicion;
        posicion *= 10;
        num /= 2;
    }

    return res;
}

// 57 - Hexadecimal a decimal
int hexadecimalDecimal(char cadena[]) {
    int res = 0;
    int longitud = strlen(cadena);

    for (int i = 0; i < longitud; i++) {

        char caracter = toupper(cadena[i]);
        int valor;

        if (caracter >= '0' && caracter <= '9') {
            valor = caracter - '0';
        } else {
            valor = caracter - 'A' + 10;
        }

        res = (res * 16) + valor;
    }

    return res;
}

// 58 - Decimal a hexadecimal
void decimalHexadecimal(int num, char cadena[]) {

    char aux[100];
    int i = 0;
    int residuo;

    while (num > 0) {

        residuo = num % 16;

        if (residuo < 10) {
            aux[i] = residuo + '0';
        } else {
            aux[i] = residuo - 10 + 'A';
        }

        i++;
        num /= 16;
    }

    aux[i] = '\0';

    int longitud = strlen(aux);

    for (int j = 0; j < longitud; j++) {
        cadena[j] = aux[longitud - 1 - j];
    }

    cadena[longitud] = '\0';
}
